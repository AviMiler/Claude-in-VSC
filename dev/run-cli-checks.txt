function Msg($t) { @{ type = 'user'; message = @{ role = 'user'; content = @(@{ type = 'text'; text = $t }) } } | ConvertTo-Json -Depth 6 -Compress }

# מריץ את ה-CLI כשהקלט בא מקובץ והפלט נכתב לקבצים. מחזיר קוד יציאה, שורות stdout ושורות stderr.
function RunClaude($lines, $extra, $workDir) {
  $in = Join-Path $env:TEMP 'cl-in.json'; $o = Join-Path $env:TEMP 'cl-out.txt'; $e = Join-Path $env:TEMP 'cl-err.txt'
  [System.IO.File]::WriteAllText($in, (($lines -join "`n") + "`n"), (New-Object System.Text.UTF8Encoding($false)))
  Remove-Item $o, $e -ErrorAction SilentlyContinue
  $argList = @('--model', $Model, '-p', '--input-format', 'stream-json', '--output-format', 'stream-json', '--verbose') + @($extra)
  $p = Start-Process -FilePath $claude -ArgumentList $argList -WorkingDirectory $workDir -RedirectStandardInput $in -RedirectStandardOutput $o -RedirectStandardError $e -NoNewWindow -Wait -PassThru
  [pscustomobject]@{
    Code = $p.ExitCode
    Out  = @(Get-Content $o -Encoding UTF8 -ErrorAction SilentlyContinue)
    Err  = @(Get-Content $e -Encoding UTF8 -ErrorAction SilentlyContinue)
  }
}

function Diag($r) {
  'קוד יציאה: ' + $r.Code + ' | שורות stdout: ' + $r.Out.Count + ' | שורות stderr: ' + $r.Err.Count
  if ($r.Err.Count) { 'stderr (עד 8 שורות):'; $r.Err | Select-Object -First 8 }
  if ($r.Out.Count -and -not ($r.Out | Where-Object { $_ -match '"type":"result"' })) { 'stdout (עד 5 שורות, בלי result):'; $r.Out | Select-Object -First 5 | ForEach-Object { if ($_.Length -gt 300) { $_.Substring(0, 300) + '...' } else { $_ } } }
}

function Ask($label, $text) {
  Write-Host ('--- ' + $label) -ForegroundColor Yellow
  $r = RunClaude @((Msg $text)) @('--disallowedTools', 'Read,Glob,Grep,Bash,Edit,Write,WebFetch') $dir
  $res = $null
  foreach ($l in $r.Out) { try { $j = $l | ConvertFrom-Json } catch { continue }; if ($j.type -eq 'result') { $res = $j } }
  if ($res) { 'תשובה: [' + $res.result + ']  (is_error=' + $res.is_error + ')' } else { 'לא התקבלה תשובה.' }
  if (-not $res -or $res.is_error) { Diag $r }
}
